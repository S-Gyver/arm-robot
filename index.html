ESP32 Firmware Guide & Wiring

📌 การต่อสาย Servo เข้ากับบอร์ด ESP32:
  • Base (ฐานหมุน): GPIO 18 (ส้ม/เหลือง)
  • Shoulder (หัวไหล่): GPIO 19 (ส้ม/เหลือง)
  • Elbow (ข้อศอก): GPIO 21 (ส้ม/เหลือง)
  • Gripper (มือจับ): GPIO 22 (ส้ม/เหลือง)
  • VCC / GND: ต่อไฟเลี้ยงแยก 5V 2A-3A (ห้ามดึงไฟจาก 3.3V ของ ESP32 โดยตรง) และต่อสาย GND ร่วมกัน
Arduino Sketch (รองรับไลบรารี ESP32Servo): Baud: 115200 
#include <ESP32Servo.h>

Servo sBase, sShoulder, sElbow, sGripper;

void setup() {
  Serial.begin(115200);
  sBase.attach(18);
  sShoulder.attach(19);
  sElbow.attach(21);
  sGripper.attach(22);
  sBase.write(90); sShoulder.write(90); sElbow.write(90); sGripper.write(90);
}

void loop() {
  if (Serial.available() > 0) {
    String str = Serial.readStringUntil('\n');
    str.trim();
    int c1 = str.indexOf(',');
    int c2 = str.indexOf(',', c1 + 1);
    int c3 = str.indexOf(',', c2 + 1);
    if (c1 > 0 && c2 > c1 && c3 > c2) {
      int b = constrain(str.substring(0, c1).toInt(), 0, 180);
      int s = constrain(str.substring(c1 + 1, c2).toInt(), 0, 180);
      int e = constrain(str.substring(c2 + 1, c3).toInt(), 0, 180);
      int g = constrain(str.substring(c3 + 1).toInt(), 0, 180);
      sBase.write(b); sShoulder.write(s); sElbow.write(e); sGripper.write(g);
      Serial.printf("ACK: %d,%d,%d,%d\n", b, s, e, g);
    }
  }
}
