ESP32 Firmware Guide & Wiring (2 - 6 Servos)

ขาต่อ GPIO ของ Servo แต่ละรุ่น:
2 Servos:
• S1 (Base): GPIO 18
• S2 (Gripper): GPIO 19
3 Servos:
• S1 (Base): GPIO 18
• S2 (Arm): GPIO 19
• S3 (Gripper): GPIO 21
4 Servos (Standard):
• S1 (Base): GPIO 18 | S2 (Shoulder): GPIO 19
• S3 (Elbow): GPIO 21 | S4 (Gripper): GPIO 22
5-6 Servos:
• S1: 18 | S2: 19 | S3: 21 | S4: 22
• S5 (Wrist): GPIO 23 | S6 (Gripper): GPIO 25

⚠️ คำแนะนำเรื่องไฟ: ควรใช้แหล่งจ่าย 5V 2A-3A ภายนอกเลี้ยงเซอร์โว และต่อสาย GND ร่วมกับ ESP32 เสมอ

Arduino Sketch (Universal Dynamic 2-6 Servos): Baud: 115200 
#include <ESP32Servo.h>

// กำหนดจำนวนเซอร์โวที่ใช้งาน (2, 3, 4, 5, หรือ 6)
#define NUM_SERVOS 4

// กำหนดขา GPIO ตามลำดับเซอร์โว
const int SERVO_PINS[6] = {18, 19, 21, 22, 23, 25};
Servo servos[6];

void setup() {
  Serial.begin(115200);
  delay(500);

  // ผูกเซอร์โวตามจำนวน NUM_SERVOS
  for (int i = 0; i < NUM_SERVOS; i++) {
    servos[i].attach(SERVO_PINS[i], 500, 2400);
    servos[i].write(90); // Home position
  }
  Serial.printf("ESP32 Ready! Active Servos: %d\n", NUM_SERVOS);
}

void loop() {
  if (Serial.available() > 0) {
    String line = Serial.readStringUntil('\n');
    line.trim();
    if (line.length() == 0) return;

    // แยกค่าองศาที่คั่นด้วยจุลภาค เช่น "90,120,45,10"
    int index = 0;
    int lastPos = 0;
    for (int i = 0; i <= line.length() && index < NUM_SERVOS; i++) {
      if (i == line.length() || line.charAt(i) == ',') {
        int angle = constrain(line.substring(lastPos, i).toInt(), 0, 180);
        servos[index].write(angle);
        index++;
        lastPos = i + 1;
      }
    }
    Serial.printf("ACK: %s\n", line.c_str());
  }
}
